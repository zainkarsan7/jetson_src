cmake_minimum_required(VERSION 3.8)
project(hb_robot_cut_environment)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(hb_robot_interfaces REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS kdtree)
add_executable(capture_cut_environment src/capture_cut_environment.cpp src/environment_filter.cpp)
target_compile_features(capture_cut_environment PRIVATE cxx_std_17)
target_include_directories(capture_cut_environment PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(capture_cut_environment Eigen3::Eigen ${PCL_LIBRARIES})
ament_target_dependencies(capture_cut_environment rclcpp tf2_ros hb_robot_interfaces
  moveit_msgs sensor_msgs geometric_shapes)
install(TARGETS capture_cut_environment DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(cut_environment_test test/test_cut_environment.py
    ENV ROS_LOCALHOST_ONLY=1 ROS_DOMAIN_ID=183 CAPTURE_EXECUTABLE=$<TARGET_FILE:capture_cut_environment>)
endif()
ament_package()
