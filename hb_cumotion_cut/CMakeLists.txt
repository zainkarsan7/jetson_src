cmake_minimum_required(VERSION 3.16)
project(hb_cumotion_cut)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
foreach(dep geometry_msgs sensor_msgs trajectory_msgs moveit_msgs)
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME}
  "srv/SolveProcessPath.srv" "srv/PlanTransit.srv"
  DEPENDENCIES geometry_msgs sensor_msgs trajectory_msgs moveit_msgs)
ament_export_dependencies(rosidl_default_runtime)
option(CONTAINER_WORKER_ONLY "Install only ROS interfaces and the GPU worker" OFF)
install(PROGRAMS scripts/gpu_worker.py DESTINATION lib/${PROJECT_NAME})
if(NOT CONTAINER_WORKER_ONLY)
  set(host_deps hb_robot_interfaces hb_robot_perception rclcpp rclcpp_action std_srvs
      std_msgs moveit_ros_planning_interface moveit_ros_planning moveit_core
      tf2_eigen visualization_msgs ur_msgs)
  foreach(dep IN LISTS host_deps)
    find_package(${dep} REQUIRED)
  endforeach()
  find_package(hb_robot_skills REQUIRED)
  # hb_robot_skills installs headers and an archive without exporting a target.
  get_filename_component(skills_prefix "${hb_robot_skills_DIR}/../../.." ABSOLUTE)
  find_library(motion_archive hb_motion PATHS "${skills_prefix}/lib" NO_DEFAULT_PATH REQUIRED)
  add_executable(cut_profile_server src/cut_profile_server.cpp src/cut_recovery.cpp
      src/gpu_process_generator.cpp src/gpu_plans.cpp)
  target_compile_features(cut_profile_server PRIVATE cxx_std_17)
  target_include_directories(cut_profile_server PRIVATE include "${skills_prefix}/include")
  ament_target_dependencies(cut_profile_server ${host_deps})
  rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
  target_link_libraries(cut_profile_server ${motion_archive} hb_robot_perception::hb_perception ${typesupport})
  install(TARGETS cut_profile_server DESTINATION lib/${PROJECT_NAME})
  install(PROGRAMS scripts/benchmark.py DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
  if(BUILD_TESTING)
    find_package(ament_cmake_gtest REQUIRED)
    find_package(urdf REQUIRED)
    find_package(srdfdom REQUIRED)
    ament_add_gtest(process_bridge_test test/process_bridge_test.cpp src/gpu_process_generator.cpp)
    target_compile_features(process_bridge_test PRIVATE cxx_std_17)
    target_include_directories(process_bridge_test PRIVATE include "${skills_prefix}/include")
    ament_target_dependencies(process_bridge_test ${host_deps} urdf srdfdom)
    target_link_libraries(process_bridge_test hb_robot_perception::hb_perception ${typesupport})
    set_tests_properties(process_bridge_test PROPERTIES ENVIRONMENT "ROS_LOCALHOST_ONLY=1;ROS_DOMAIN_ID=187")
  endif()
endif()
ament_package()
